Q1. Accept n numbers in an array and display the elements (5 marks)
Program key: array display. REPEATED 2 times — Slip 5 Q1, Slip 27 Q1
#include <stdio.h>
int main(void) {
    int a[100], n, i;
    printf("Enter number of elements: "); scanf("%d", &n);
    if (n < 1 || n > 100) return 0;
    for (i = 0; i < n; i++) { printf("Element: "); scanf("%d", &a[i]); }
    printf("Array elements: ");
    for (i = 0; i < n; i++) printf("%d ", a[i]);
    printf("\n");
    return 0;
}
Q2. Display the multiplication table of n (10 marks)
Program key: table. REPEATED 3 times — Slip 5 Q2, Slip 18 Q2, Slip 22 Q2
#include <stdio.h>
int main(void) {
    int n, i;
    printf("Enter number: "); scanf("%d", &n);
    for (i = 1; i <= 10; i++)
        printf("%d x %d = %d\n", n, i, n * i);
    return 0;
}
Q3. Input radius; menu for area of circle, circumference of circle, volume of sphere (10 marks)
Program key: circle sphere menu. REPEATED 3 times — Slip 5 Q3, Slip 15 Q3, Slip 18 Q3
#include <stdio.h>
#define PI 3.141592653589793
int main(void) {
    double r;
    int choice;
    printf("Enter radius: "); scanf("%lf", &r);
    printf("1.Area of circle 2.Circumference 3.Volume of sphere\nChoice: ");
    scanf("%d", &choice);
    switch (choice) {
        case 1: printf("Area = %.2f\n", PI * r * r); break;
        case 2: printf("Circumference = %.2f\n", 2 * PI * r); break;
        case 3: printf("Volume = %.2f\n", (4.0 / 3.0) * PI * r * r * r); break;
        default: printf("Invalid choice\n");
    }
    return 0;
}
